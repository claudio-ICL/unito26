\begin{frame}
  \frametitle{The idealised book}
\begin{assumption}
\label{assumption.idealisedBook}
 For a fixed $\uniformDepth > 0$:
 \begin{enumerate}[label={\emph{\roman{*}}.}]
  \item no event occurs behind the best quote of its side;
  \item no order reaches past the quote it acts on;
  \item every queue holds at most $\uniformDepth$, and every queue behind a touch exactly
        $\uniformDepth$;
  \item at a full best bid, the next bid limit order arrives one tick above it.
 \end{enumerate}
\end{assumption}
\note{
Part iv is written for the bid; for the ask, the next ask limit order arrives one tick below
a full best ask.
Parts i, iii and iv are the stylised book of \cite{CKS14pri}. Part ii is not: there the flows
are totalled at whichever quote is best when each order arrives, so an order reaching past the
touch is split across the levels it crosses. We ask instead that the contribution of an event
be the size of that event, and part ii is what that costs.
}
\end{frame}

\begin{frame}
  \frametitle{The order flow imbalance}
\begin{defi}[\citealp{CKS14pri}]
\label{def.orderFlowImbalance}
 \[
 \begin{aligned}
  \orderFlowContribution
  = {}& \indicator{\bestBidPrice_{\arrivalTimes[]_n} \geq \bestBidPrice_{\arrivalTimes[]_{n-1}}}
  \nthBestBidSize[1]_{\arrivalTimes[]_n}
  - \indicator{\bestBidPrice_{\arrivalTimes[]_n} \leq \bestBidPrice_{\arrivalTimes[]_{n-1}}}
  \nthBestBidSize[1]_{\arrivalTimes[]_{n-1}} \\
  & - \indicator{\bestAskPrice_{\arrivalTimes[]_n} \leq \bestAskPrice_{\arrivalTimes[]_{n-1}}}
  \nthBestAskSize[1]_{\arrivalTimes[]_n}
  + \indicator{\bestAskPrice_{\arrivalTimes[]_n} \geq \bestAskPrice_{\arrivalTimes[]_{n-1}}}
  \nthBestAskSize[1]_{\arrivalTimes[]_{n-1}},
 \end{aligned}
 \]
 and $\OFI_{t,\ofiWindow} = \sum_{t-\ofiWindow < \arrivalTimes[]_n \leq t}
 \orderFlowContribution$.
\end{defi}

\pause

$\nthBestBidSize[1]_{\arrivalTimes[]_n}$: the size at the best bid \emph{after} the $n$-th
event
\note{
The convention is the one that is got wrong: the size at the best bid after the $n$-th event,
so that $\nthBestBidSize[1]_{\arrivalTimes[]_{n-1}}$ is the size standing just before it.
Both bid indicators fire when the best bid does not move, so an event that leaves it in place
contributes the change in the size resting there. An event that moves it contributes a whole
queue.
}

\begin{frame}
  \frametitle{The mid-price moves with the flow}
\begin{prop}
\label{prop.idealisedUpdate}
 Under the idealised book,
 $\Delta\midPrice_{t,\ofiWindow}
 = \frac{\tickSizeOfLOB}{2\uniformDepth}\OFI_{t,\ofiWindow} + \residual_{t,\ofiWindow}$,
 with $\vert\residual_{t,\ofiWindow}\vert \leq \tickSizeOfLOB$.
\end{prop}

\pause

$\residual_{t,\ofiWindow} = -\frac{\tickSizeOfLOB}{2\uniformDepth}\big[
(\nthBestBidSize[1]_t - \nthBestBidSize[1]_{t-\ofiWindow})
- (\nthBestAskSize[1]_t - \nthBestAskSize[1]_{t-\ofiWindow}) \big]$

\pause

Exact when the two touches change by the same amount
\note{
The residual depends only on the sizes resting at the two touches at the ends of the window.
In particular the identity is exact whenever the bid and the ask touches change by the same
amount over the window, and so whenever all four sizes equal $\uniformDepth$.
The proof is the update rule of section 1.1 specialised to the assumption, and it is the
exercise on the next frame.
}
\end{frame}

\begin{remark}
\label{remark.residualIsQueueFlow}
 If neither quote moves over the window,
 $\residual_{t,\ofiWindow} = -\tickSizeOfLOB\,\OFI_{t,\ofiWindow}/(2\uniformDepth)$.
\end{remark}
\note{
Each dot is a window. The shaded band is the residual's bound, one tick either side of the
line. The orange dots are windows over which neither quote moved: the mid-price change is zero
while the imbalance is not, and the two terms of the identity cancel exactly.
The residual is the part of the flow that went into the queues rather than into the quotes. It
is bounded by one tick and does not grow with $\ofiWindow$, which is what makes the identity
useful over windows whose flow term is larger than that.
}
\end{frame}

\begin{defi}
\label{def.orderFlowProcess}
 $(\arrivalTimes[]_n, \event[n])$ an exponential-kernel Hawkes process of dimension
 $6$; sizes $\orderSize_n$ i.i.d.\ with mean $\meanSize$, independent of both.
\end{defi}
\note{
The six types are the market, limit and withdrawal orders on each side. The vector $\pressure$
records the pressure each type exerts on the mid-price: a market buy consumes the ask and a
withdrawal from the bid removes support, so the first pushes up and the second down.
A specification of the order flow is a triple $(\baseIntensity, \excitation, \decay)$ for the
Hawkes process, exactly as in the last class.
}
\end{frame}

behind the best bid: probability
$(1-\offsetParameter)^{\LOBspread/\tickSizeOfLOB}$
\note{
A limit buy. The lower row is the book, the upper row the law of the price. At offset zero
the order is one tick from the opposite best quote: it improves the best quote of its side
while the spread is wider than a tick, and joins it at a spread of one tick. From the offset
$\LOBspread/\tickSizeOfLOB$ on it arrives behind its best quote, where it moves neither quote
and contributes nothing to the imbalance. A market order needs no price. A withdrawal draws
an occupied price of its side in proportion to the size resting there, and cancels its size
or what rests there, whichever is less.
The idealised book asks for no arrival behind the quotes and for a full queue on every tick
behind them: it takes that depth as given. A book generated by the flow holds behind its
quotes only what the flow has left there, so the parameter stays below one, and the next
statement holds for the events at the best quotes only.
The frame of the rule is Cont, Stoikov and Talreja's. They fit a power law, and observed
placement is heavy-tailed where this law is not.
}

\begin{frame}
  \frametitle{The contribution is the signed size}
\begin{prop}
\label{prop.signedSizeContribution}
 Under the idealised book, $\orderFlowContribution = \pressure_{\event[n]}\orderSize_n$, so
 $\OFI_{t,\ofiWindow} = \sum_{t-\ofiWindow < \arrivalTimes[]_n \leq t}
 \pressure_{\event[n]}\orderSize_n$.
\end{prop}

\pause

Part ii: each event changes one queue, by its own size
\note{
The first two terms of the contribution are the signed flow on the bid side, the last two
minus the signed flow on the ask side. No order reaches past the quote it acts on, so each
event changes one queue by its own size: a limit buy adds its size to the bid, a withdrawal
from the bid removes it, a market buy removes it from the ask; in each case with the sign
$\pressure_{\event[n]}$.
On a general book an order that reaches past the touch registers only what the touch held.
Now to notebook 07: the idealised book made real, what each event contributes, and the
identity checked to the last digit on the package's own session.
}
\end{frame}

\begin{frame}

\begin{defi}
\label{def.directionSymmetry}
 With $\swapMatrix$ the permutation exchanging the two members of each pair, the
 specification is \emph{direction-symmetric} if
 $\swapMatrix\excitation\swapMatrix = \excitation$ and
 $\swapMatrix\baseIntensity = \baseIntensity$.
\end{defi}

\pause

$\swapMatrix\pressure = -\pressure$

\pause

A hypothesis on the kernel, not only on the baseline

Swapping buy for sell leaves the specification alone. This is the hypothesis every forecast of

\begin{frame}
  \frametitle{Contracted with $\pressure$, the baseline drops out}
\begin{prop}
\label{prop.directionSymmetry}
 Under direction symmetry, $\pressure^{\transpose}Mv = 0$ whenever
 $\swapMatrix M\swapMatrix = M$ and $\swapMatrix v = v$. In particular
 $\pressure^{\transpose}\baseIntensity = 0$, and $\pressure^{\transpose}\stationaryIntensity =
 0$ if $\branchingRatio < 1$.
\end{prop}

\note{
$\pressure^{\transpose}Mv = (\swapMatrix\pressure)^{\transpose}(\swapMatrix M\swapMatrix)
(\swapMatrix v)\cdot(-1)$, which is minus itself.
The matrices fixed by $M \mapsto \swapMatrix M\swapMatrix$ form a closed subspace, closed under
products, so they contain $\excitation$ and $\hurwitzMatrix$ together with the limits and
integrals of their polynomials. Every expression of the forecast built from these and
contracted with $\pressure$ therefore loses its baseline term.
}

\begin{frame}
  \frametitle{The intensity contrast}
\begin{defi}
\label{def.intensityContrast}
 $\intensityContrast(t) := \upIntensity(t) - \downIntensity(t)
 = \pressure^{\transpose}\intensity[ ](t)
 = \pressure^{\transpose}\baseIntensity + \pressure^{\transpose}\excitation\decayedCounts(t)$.
\end{defi}

\pause

Under direction symmetry: $\intensityContrast(t) =
\pressure^{\transpose}\excitation\decayedCounts(t)$, a function of the state
\note{
$\upIntensity$ sums the intensities of the three types that push the mid-price up, and
$\downIntensity$ those of the three that push it down. The contrast says in which direction
the flow is currently leaning.
Under direction symmetry the baseline term vanishes, and the contrast is a function of the
state alone.
}
\end{frame}

\begin{frame}
  \frametitle{Signed excitation}
\begin{prop}
\label{prop.signedExcitation}
 Under direction symmetry,
 $\signedExcitation_{\eone} := (\pressure^{\transpose}\excitation)_{\eone}/\pressure_{\eone}$
 takes one value on each pair, and
 $\intensityContrast(t) = \sum_{\eone}\pressure_{\eone}\signedExcitation_{\eone}
 \decayedCounts_{\eone}(t)$.
\end{prop}

\pause

$\signedExcitation > 0$: begets its own pressure. $\signedExcitation < 0$: the opposite.

\pause

A property of the kernel, not of the type
\note{
One event of type $\eone$ raises the intensity of the types that push its own way, relative to
those that push the other way, by $\signedExcitation_{\eone}$. A type with a positive value
begets its own pressure, one with a negative value begets the opposite, and one with zero
begets neither. The same six types carry either sign under different specifications.

\begin{frame}
  \frametitle{The short-horizon forecast}
\begin{prop}
\label{prop.shortHorizonForecast}
 For a direction-symmetric specification and every $\forecastHorizon \geq 0$,
 \[
  \Expectation\Big[ \sum_{t < \arrivalTimes[]_n \leq t+\forecastHorizon}
  \pressure_{\event[n]}\orderSize_n \,\Big|\, \sigmaAlgebra_t \Big]
  = \meanSize\,\pressure^{\transpose}\excitation\,
  \meanResponseIntegral(\forecastHorizon)\,\decayedCounts(t+).
 \]
\end{prop}

\pause

$\meanResponseIntegral(\forecastHorizon) = \int_0^{\forecastHorizon}
e^{\hurwitzMatrix s}ds$, from the forward mean of 1.3

\pause

Under the idealised book, the left side is $\OFI_{t+\forecastHorizon,\forecastHorizon}$
\note{
The frame the chapter exists for. The marks are independent of the arrival times and the
types, with mean $\meanSize$, so the left side is $\meanSize$ times the pressure-weighted
expected count over the horizon. By the forward mean of the last class that expected count is
$\forecastHorizon\baseIntensity + \excitation\meanResponseIntegral(\forecastHorizon)
\decayedCounts(t+) + \excitation(\int_0^{\forecastHorizon}\meanResponseIntegral)
\baseIntensity$.
The first and third terms are $\swapMatrix$-invariant matrices applied to the baseline, and
$\pressure^{\transpose}$ annihilates them. What remains is the forecast.
$\decayedCounts(t+)$ is the state read \emph{after} the event at $t$, if there is one.
}
\end{frame}

\begin{frame}
  \frametitle{The forecast finishes before the relaxation time}
\begin{remark}
\label{remark.totalResponse}
 $\excitation\meanResponseIntegral(\infty) = \branchingMatrix(I-\branchingMatrix)^{-1}$, and
 $\pressure^{\transpose}\excitation u = 0$ for the right Perron vector $u$.
\end{remark}

\pause

The timescale: the slowest mode that survives the contraction with $\pressure$
\note{
At the other end, $\excitation\meanResponseIntegral(\infty)$ is the whole expected progeny of
the present state: direct offspring composed with all descendants. So the state-dependent part
of the forecast has finished at horizons long against the relaxation time.
It finishes strictly sooner. Let $\branchingMatrix$ be irreducible with right Perron vector
$u$. Direction symmetry makes $\swapMatrix u$ a Perron vector as well, and simplicity with
$u > 0$ forces $\swapMatrix u = u$, so that
$\pressure^{\transpose}\excitation u = \decay\branchingRatio\,\pressure^{\transpose}u = 0$.
The slowest mode is absent from the forecast at every horizon. On the two shipped regimes the
surviving rate differs from pair to pair, and the notebook reads it off.
}

\begin{corol}
\label{corol.regimeSign}
 An extra event of type $a$, with $\pressure_a = +1$, against one fewer of its mirror moves
 the forecast by $2\meanSize\signedExcitation_a(\forecastHorizon)$, where
 $\signedExcitation_{\eone}(\forecastHorizon) := (\pressure^{\transpose}\excitation
 \meanResponseIntegral(\forecastHorizon))_{\eone}/\pressure_{\eone}$.
\end{corol}
\note{
$\signedExcitation(\forecastHorizon)$ takes one value on each pair, and
$\signedExcitation(\forecastHorizon)/\forecastHorizon \to \signedExcitation$ as the horizon
shrinks. Its sign decides whether an excess of type $a$ over its mirror is forecast to push
the flow further in its own direction or to reverse it, and that sign belongs to the kernel.
Two specifications with the same branching ratio and the same stationary intensity can carry
opposite signs, and then forecast opposite displacements from such a state, while the identity
of the idealised book holds identically in both. At a general state the forecast is
$\meanSize\sum_{\eone}\pressure_{\eone}\signedExcitation_{\eone}(\forecastHorizon)
\decayedCounts_{\eone}(t+)$, and two specifications of opposite sign need not disagree on it.
No statement of the form ``the order flow imbalance predicts continuation'' is made without
naming the specification it is made in.
}
