Proposition \ref{prop.idealisedUpdate} is an identity about a window that has elapsed.
This subsection asks the forward question:
given everything observed up to $t$, where does the flow take the mid-price over
$(t, t+\forecastHorizon]$.

Throughout, $\sigmaAlgebra_t$ is the history of the order flow up to and including $t$,
and $\decayedCounts(t+)$ is the state read after the event at $t$, if there is one.

\begin{prop}\label{prop.shortHorizonForecast}
 Let the order flow be as in Definition \ref{def.orderFlowProcess},
 with a direction-symmetric specification,
 and write
 $\meanResponseIntegral(\forecastHorizon)
 := \int_{0}^{\forecastHorizon} e^{\hurwitzMatrix s} \, ds$
 for $\hurwitzMatrix = \excitation - \decay I$.
 Then for every $\forecastHorizon \geq 0$
 \begin{equation}\label{eq.shortHorizonForecast}
  \Expectation \Big[ \sum_{n \, : \, t < \arrivalTimes[]_n \leq t+\forecastHorizon}
  \pressure_{\event[n]} \orderSize_n \,\Big|\, \sigmaAlgebra_t \Big]
  = \meanSize \; \pressure^{\transpose} \excitation \,
  \meanResponseIntegral(\forecastHorizon) \, \decayedCounts(t+) .
 \end{equation}
 Under Assumption \ref{assumption.idealisedBook} the sum on the left is
 $\OFI_{t+\forecastHorizon, \forecastHorizon}$,
 by Proposition \ref{prop.signedSizeContribution}.
\end{prop}
\begin{proof}
 The marks are independent of the arrival times and of the types, with mean $\meanSize$,
 so the left-hand side is
 $\meanSize \, \pressure^{\transpose}
 \Expectation[\multiCountingProc(t+\forecastHorizon) - \multiCountingProc(t)
 \mid \sigmaAlgebra_t]$,
 the sum being absolutely summable because the expected count is finite.
 By Theorem \ref{thm.markovState} and Proposition \ref{prop.forwardMean}, that expectation is
 \begin{equation*}
  \forecastHorizon \, \baseIntensity
  + \excitation \, \meanResponseIntegral(\forecastHorizon) \, \decayedCounts(t+)
  + \excitation \Big( \int_{0}^{\forecastHorizon} \meanResponseIntegral(s) \, ds \Big)
  \baseIntensity .
 \end{equation*}
 The first and third terms are $\swapMatrix$-invariant matrices applied to $\baseIntensity$,
 so $\pressure^{\transpose}$ annihilates them by Proposition \ref{prop.directionSymmetry},
 and \eqref{eq.shortHorizonForecast} remains.
\end{proof}

\begin{corol}\label{corol.priceForecast}
 Let the order flow be as in Definition \ref{def.orderFlowProcess},
 with a direction-symmetric specification,
 and let Assumption \ref{assumption.idealisedBook} be in force.
 Then the time-$t$ expectation of the mid-price change over
 $(t, t+\forecastHorizon]$ is
 \begin{equation}\label{eq.priceForecast}
  \Expectation \big[ \midPrice_{t+\forecastHorizon} - \midPrice_{t}
  \,\big|\, \sigmaAlgebra_t \big]
  = \frac{\tickSizeOfLOB \meanSize}{2\uniformDepth} \;
  \pressure^{\transpose} \excitation \, \meanResponseIntegral(\forecastHorizon) \,
  \decayedCounts(t+)
  + \Expectation \big[ \residual_{t+\forecastHorizon, \forecastHorizon}
  \,\big|\, \sigmaAlgebra_t \big] ,
 \end{equation}
 with $\vert \residual_{t+\forecastHorizon, \forecastHorizon} \vert
 \leq \tickSizeOfLOB$.
\end{corol}
\begin{proof}
 Take conditional expectations in \eqref{eq.idealisedUpdate} over the window
 $(t, t+\forecastHorizon]$ and apply \eqref{eq.shortHorizonForecast}.
\end{proof}

\begin{remark}\label{remark.residualAgainstForecast}
 The residual is carried rather than absorbed, for two reasons.
 By Remark \ref{remark.residualIsQueueFlow} it is not independent of the first term:
 along a path on which neither quote moves the two cancel exactly.
 And it does not shrink as the forecast grows:
 $\meanResponseIntegral(\forecastHorizon) \to -\hurwitzMatrix^{-1}$ as
 $\forecastHorizon \to \infty$ when $\branchingRatio < 1$,
 so the first term of \eqref{eq.priceForecast} is bounded in $\forecastHorizon$,
 while the residual is bounded by one tick at every horizon.
 A long horizon therefore does not buy a forecast that exceeds the residual:
 \eqref{eq.priceForecast} carries information about the mid-price only where
 $\meanSize \,\vert \pressure^{\transpose}\branchingMatrix (I - \branchingMatrix)^{-1}
 \decayedCounts(t+) \vert$ is itself of the order of $\uniformDepth$.
 Section \ref{sec.lobsterEmpirics} asks the corresponding empirical question about the sign
 of \eqref{eq.priceForecast}, and not about its magnitude.
\end{remark}

\begin{corol}\label{corol.contrastAsDrift}
 Let the order flow be as in Definition \ref{def.orderFlowProcess},
 with a direction-symmetric specification,
 and let Assumption \ref{assumption.idealisedBook} be in force.
 Then
 \begin{equation}\label{eq.contrastAsDrift}
  \Expectation \big[ \OFI_{t+\forecastHorizon, \forecastHorizon}
  \,\big|\, \sigmaAlgebra_t \big]
  = \meanSize \, \intensityContrast(t+) \, \forecastHorizon
  + O(\forecastHorizon^{2})
 \end{equation}
 as $\forecastHorizon \to 0$.
\end{corol}
\begin{proof}
 $\meanResponseIntegral(\forecastHorizon)
 = \forecastHorizon I + \tfrac{1}{2}\forecastHorizon^{2}\hurwitzMatrix
 + O(\forecastHorizon^{3})$,
 and $\pressure^{\transpose}\excitation\decayedCounts(t+) = \intensityContrast(t+)$ by
 \eqref{eq.contrastFromState}.
\end{proof}

Up to the mean size, the intensity contrast is therefore the drift of the order flow
imbalance, to leading order in the horizon and about the flow,
the price inheriting it only through \eqref{eq.priceForecast}.

\begin{remark}\label{remark.totalResponse}
 At the other end, if $\branchingRatio < 1$ then $\hurwitzMatrix$ is Hurwitz by
 Proposition \ref{prop.hurwitzCriterion} and
 $\excitation \meanResponseIntegral(\infty)
 = -\excitation\hurwitzMatrix^{-1}
 = \branchingMatrix (I - \branchingMatrix)^{-1}$,
 which is the whole expected progeny of the present state:
 direct offspring composed with all descendants.
 The state-dependent part of \eqref{eq.shortHorizonForecast} has therefore finished at
 horizons long against the relaxation time \eqref{eq.relaxationTime}.

 It finishes strictly sooner.
 Let the specification be direction-symmetric,
 and let $\branchingMatrix$ be irreducible, with right Perron vector $u$.
 Then \eqref{eq.directionSymmetry} makes $\swapMatrix u$ a Perron vector as well,
 and simplicity together with $u > 0$ forces $\swapMatrix u = u$, so that
 $\pressure^{\transpose}\excitation u
 = \decay\branchingRatio\,\pressure^{\transpose}u = 0$
 by Proposition \ref{prop.directionSymmetry}.
 The slowest mode is absent from \eqref{eq.shortHorizonForecast} at every horizon,
 and the timescale is that of the slowest mode that survives the contraction with
 $\pressure^{\transpose}$.
\end{remark}

\begin{corol}\label{corol.regimeSign}
 Let the order flow be as in Definition \ref{def.orderFlowProcess},
 with a direction-symmetric specification, and set
 \begin{equation}\label{eq.signedExcitationAtHorizon}
  \signedExcitation_{\eone}(\forecastHorizon)
  := \frac{\big( \pressure^{\transpose}\excitation
  \meanResponseIntegral(\forecastHorizon) \big)_{\eone}}{\pressure_{\eone}} .
 \end{equation}
 Then $\signedExcitation(\forecastHorizon)$ takes the same value on the two members of each
 pair, and for a state carrying one extra event of a type $a$ with $\pressure_a = +1$
 against one fewer of its mirror,
 the forecast \eqref{eq.shortHorizonForecast} moves by
 $2\meanSize\signedExcitation_{a}(\forecastHorizon)$.
 Moreover $\signedExcitation(\forecastHorizon)/\forecastHorizon \to \signedExcitation$ as
 $\forecastHorizon \to 0$.
\end{corol}
\begin{proof}
 $\excitation\meanResponseIntegral(\forecastHorizon)$ is $\swapMatrix$-invariant,
 so the argument of Proposition \ref{prop.signedExcitation} applies verbatim.
 The displacement of \eqref{eq.shortHorizonForecast} is
 $\meanSize\,\pressure^{\transpose}\excitation\meanResponseIntegral(\forecastHorizon)
 (\unitVector_a - \unitVector_{\swapMatrix(a)})
 = \meanSize \big( \pressure_a \signedExcitation_a(\forecastHorizon)
 - \pressure_{\swapMatrix(a)} \signedExcitation_{\swapMatrix(a)}(\forecastHorizon) \big)
 = 2\meanSize\,\pressure_a\signedExcitation_a(\forecastHorizon)$,
 which is $2\meanSize\signedExcitation_a(\forecastHorizon)$ at $\pressure_a = +1$.
 The last claim is $\meanResponseIntegral(\forecastHorizon)
 = \forecastHorizon I + O(\forecastHorizon^{2})$.
\end{proof}

The sign of $\signedExcitation_{a}(\forecastHorizon)$ therefore decides whether an excess of
type $a$ over its mirror is forecast to push the flow further in its own direction or to
reverse it, and that sign belongs to the kernel.
Two specifications with the same branching ratio and the same stationary intensity can carry
$\signedExcitation$ of opposite sign, and then they forecast opposite displacements from
such a state at every horizon at which the signs persist,
while Proposition \ref{prop.idealisedUpdate} holds identically in both.
At a general state the forecast is
$\meanSize\sum_{\eone}\pressure_{\eone}\signedExcitation_{\eone}(\forecastHorizon)
\decayedCounts_{\eone}(t+)$,
and two specifications of opposite sign need not disagree on it.
No statement of the form ``the order flow imbalance predicts continuation'' is made in these
notes without naming the specification it is made in.

\begin{remark}\label{remark.imbalanceAsProxy}
 Three statistics of the same past flow have appeared, and they differ in two ways.
 The signed count $\pressure^{\transpose}
 \big( \multiCountingProc(t) - \multiCountingProc(t-\ofiWindow) \big)$
 weights the events of a window equally and ignores their sizes;
 the exponentially weighted count $\pressure^{\transpose}\decayedCounts(t)$
 weights them by age and ignores their sizes;
 and $\OFI_{t,\ofiWindow}$ weights them equally and by their sizes.
 The forecast \eqref{eq.shortHorizonForecast} of a direction-symmetric specification
 is driven by none of the three,
 but by $\pressure^{\transpose}\excitation
 \meanResponseIntegral(\forecastHorizon)\decayedCounts(t+)$,
 which weights them by age and by $\signedExcitation(\forecastHorizon)$.

 The gap in the weighting closes exactly when $\signedExcitation$ is constant across the six
 types, that is when $\pressure$ is a left eigenvector of $\excitation$:
 then $\pressure^{\transpose}\excitation\meanResponseIntegral(\forecastHorizon)$ is a
 multiple of $\pressure^{\transpose}$ at every horizon,
 and the forecast is a multiple of $\pressure^{\transpose}\decayedCounts(t+)$.
 The gap in the marks does not close at all,
 since no function of $\decayedCounts$ recovers a size.
 So $\OFI$ is a statistic of the flow that needs no parameters and estimates the forecast
 only up to those two gaps, and how much is lost to each is a question about a particular
 specification.
\end{remark}

\begin{prop}\label{prop.forecastAtLongHorizons}
 Let the order flow be as in Definition \ref{def.orderFlowProcess},
 with a direction-symmetric specification with $\branchingRatio < 1$,
 in its stationary version,
 and let the sizes have finite variance.
 Write
 \begin{equation}\label{eq.signedFlow}
  \signedFlow_{t,\forecastHorizon}
  := \sum_{n \, : \, t < \arrivalTimes[]_n \leq t+\forecastHorizon}
  \pressure_{\event[n]} \orderSize_n
 \end{equation}
 for the signed flow forecast by \eqref{eq.shortHorizonForecast}, and set
 \begin{equation}\label{eq.flowVolatility}
  \countCovarianceRate
  := (I - \branchingMatrix)^{-1} \,
  \mathrm{diag}(\stationaryIntensity) \,
  (I - \branchingMatrix)^{-\transpose} ,
  \qquad
  \flowVolatility^{2}
  := \meanSize^{2} \, \pressure^{\transpose} \countCovarianceRate \, \pressure
  + \Variance(\orderSize) \, \totalRate .
 \end{equation}
 Then, as $\forecastHorizon \to \infty$,
 \begin{enumerate}[label={\emph{\roman{*}}.}]
  \item $\Variance(\signedFlow_{t,\forecastHorizon})
        = \flowVolatility^{2} \, \forecastHorizon + O(\sqrt{\forecastHorizon})$;
  \item $\Variance\big( \Expectation[ \signedFlow_{t,\forecastHorizon}
        \mid \sigmaAlgebra_t ] \big)$ is bounded in $\forecastHorizon$.
 \end{enumerate}
 The forecast therefore accounts for a share of the variance of the flow that is
 $O(1/\forecastHorizon)$.
 Under Assumption \ref{assumption.idealisedBook} the same holds of the mid-price:
 $\Variance(\midPrice_{t+\forecastHorizon} - \midPrice_{t})
 = (\tickSizeOfLOB / 2\uniformDepth)^{2} \, \flowVolatility^{2} \, \forecastHorizon
 + O(\sqrt{\forecastHorizon})$,
 and the variance of its forecast \eqref{eq.priceForecast} is bounded in $\forecastHorizon$.
\end{prop}
\begin{proof}
 Let $\martingale = \multiCountingProc - \compensator$.
 By \eqref{eq.stateDynamics} and \eqref{eq.intensityFromState},
 $d\decayedCounts = (\hurwitzMatrix\decayedCounts + \baseIntensity) \, dt + d\martingale$.
 Almost surely no event falls at $t$ or at $t+\forecastHorizon$,
 so integrating this equation and \eqref{eq.stateDynamics} over
 $(t, t+\forecastHorizon]$ gives
 \begin{equation*}
 \begin{aligned}
  \decayedCounts(t+\forecastHorizon) - \decayedCounts(t)
  & = \hurwitzMatrix \int_{t}^{t+\forecastHorizon} \decayedCounts(s) \, ds
  + \forecastHorizon \, \baseIntensity
  + \martingale(t+\forecastHorizon) - \martingale(t) , \\
  \multiCountingProc(t+\forecastHorizon) - \multiCountingProc(t)
  & = \decayedCounts(t+\forecastHorizon) - \decayedCounts(t)
  + \decay \int_{t}^{t+\forecastHorizon} \decayedCounts(s) \, ds .
 \end{aligned}
 \end{equation*}
 Eliminating the integral, with $\hurwitzMatrix = -\decay(I - \branchingMatrix)$ and
 \eqref{eq.stationaryIntensity},
 \begin{equation}\label{eq.countIncrementIdentity}
  \multiCountingProc(t+\forecastHorizon) - \multiCountingProc(t)
  = \forecastHorizon \, \stationaryIntensity
  + (I - \branchingMatrix)^{-1}
  \big( \martingale(t+\forecastHorizon) - \martingale(t) \big)
  - \branchingMatrix (I - \branchingMatrix)^{-1}
  \big( \decayedCounts(t+\forecastHorizon) - \decayedCounts(t) \big) .
 \end{equation}
 By \eqref{eq.noCommonJumps} no two components of $\martingale$ jump together,
 so their increments are uncorrelated,
 and the increment of $\martingale_e$ has variance
 $\Expectation[\compensator_e(t+\forecastHorizon) - \compensator_e(t)]
 = \stationaryIntensity[e] \, \forecastHorizon$:
 the martingale term of \eqref{eq.countIncrementIdentity} has covariance
 $\forecastHorizon \, \countCovarianceRate$.
 The state term has covariance bounded in $\forecastHorizon$, by $4$ times that of
 $\branchingMatrix (I - \branchingMatrix)^{-1} \decayedCounts(t)$,
 which is finite by Theorem \ref{thm.lyapunov}.
 By the Cauchy--Schwarz inequality the covariance of the two terms is
 $O(\sqrt{\forecastHorizon})$, hence
 $\Variance\big( \pressure^{\transpose}
 ( \multiCountingProc(t+\forecastHorizon) - \multiCountingProc(t) ) \big)
 = \forecastHorizon \, \pressure^{\transpose}\countCovarianceRate\,\pressure
 + O(\sqrt{\forecastHorizon})$.
 Given the arrival times and the types,
 $\signedFlow_{t,\forecastHorizon}$ has mean
 $\meanSize \, \pressure^{\transpose}
 ( \multiCountingProc(t+\forecastHorizon) - \multiCountingProc(t) )$
 and variance $\Variance(\orderSize)$ times the number of events in
 $(t, t+\forecastHorizon]$, whose expectation is $\totalRate \, \forecastHorizon$.
 This is \emph{i.}

 By Proposition \ref{prop.shortHorizonForecast} the forecast is
 $\meanSize \, \pressure^{\transpose} \excitation \,
 \meanResponseIntegral(\forecastHorizon) \, \decayedCounts(t)$,
 a linear function of the state with coefficients that converge as
 $\forecastHorizon \to \infty$, by Remark \ref{remark.totalResponse},
 and the state has the finite covariance $V$ of \eqref{eq.stationaryCovariance}.
 This is \emph{ii.}

 Under Assumption \ref{assumption.idealisedBook},
 $\midPrice_{t+\forecastHorizon} - \midPrice_{t}
 = (\tickSizeOfLOB / 2\uniformDepth) \, \signedFlow_{t,\forecastHorizon}
 + \residual_{t+\forecastHorizon,\forecastHorizon}$
 by Propositions \ref{prop.idealisedUpdate} and \ref{prop.signedSizeContribution},
 with a residual bounded by $\tickSizeOfLOB$,
 and both statements pass to the mid-price.
\end{proof}

\begin{remark}\label{remark.threeRangesOfHorizon}
 The forecast \eqref{eq.priceForecast} is informative about the mid-price over a range of
 horizons bounded on both sides.
 At horizons short against the time a best quote takes to move,
 the flow goes into the queues and the residual cancels it,
 by Remark \ref{remark.residualIsQueueFlow}.
 At horizons of the order of the timescale of Remark \ref{remark.totalResponse},
 the mid-price has inherited the forecast.
 At horizons long against that timescale the forecast has finished,
 while by Proposition \ref{prop.forecastAtLongHorizons} its error has a standard deviation
 of order
 $(\tickSizeOfLOB / 2\uniformDepth) \, \flowVolatility \sqrt{\forecastHorizon}$:
 the state at $t$ says nothing about the events of the following hour
 beyond what it says about those of the following seconds.

 The constant $(\tickSizeOfLOB / 2\uniformDepth) \, \flowVolatility$ is the volatility of the
 mid-price of the idealised book, and it belongs to the kernel as the sign of the forecast
 does.
 When $\pressure$ is a left eigenvector of $\excitation$,
 with $\pressure^{\transpose}\excitation = \signedExcitation \, \pressure^{\transpose}$
 as in Remark \ref{remark.imbalanceAsProxy},
 \begin{equation}\label{eq.flowVolatilityAtConstantSignedExcitation}
  \pressure^{\transpose} \countCovarianceRate \, \pressure
  = \frac{\totalRate}{(1 - \signedExcitation/\decay)^{2}} .
 \end{equation}
 A flow without excitation has $\pressure^{\transpose}\countCovarianceRate\,\pressure
 = \totalRate$.
 At the same total rate,
 a kernel whose events beget their own pressure raises the volatility above that value,
 and one whose events beget the opposite pressure lowers it.
\end{remark}
