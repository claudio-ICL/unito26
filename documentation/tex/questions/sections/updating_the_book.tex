\begin{question}
\label{question.precomputedMarketPartOnAWalkingSell}
Two functions process an incoming order on an \codehl{AggregateBook}
and return its fills as pairs \codehl{(price, size)}.
They use four methods of the book:
\codehl{best\_price(d)}, the best price of the side \codehl{d},
or \codehl{None} when that side is empty;
\codehl{size\_at(d, x)}, the size at the price \codehl{x} on the side \codehl{d},
zero where nothing rests;
\codehl{set\_size(d, x, s)}, which writes the size \codehl{s} at \codehl{x}
and removes \codehl{x} when \codehl{s} is zero;
and \codehl{levels\_map(d)}, the dictionary from price to size that holds the side \codehl{d}.
A market order carries the sentinel price of Section~\ref{notes:sec.foldImplementation},
0 for a sell and \codehl{sys.maxsize} for a buy,
which \codehl{is\_market\_price} recognises.
Alternative~A is the update of the package written as a function,
and calls the static method quoted below;
Alternative~B computes $\marketOrderSize$ first,
as in Proposition~\ref{notes:prop.decompositionOfLimitOrder}.
\begin{lstlisting}[caption={\protect\codehl{unito26.lob.orderbook.AggregateBook.resting\_price}, its docstring elided.}, label=listing.precomputedMarketPartRestingPrice]
@staticmethod
def resting_price(limit_price: int, last_fill_price: int | None) -> int | None:
    # ...
    if not is_market_price(limit_price):
        return limit_price
    return last_fill_price
\end{lstlisting}
\begin{lstlisting}[caption={Alternative A}, label=listing.precomputedMarketPartOnAWalkingSellA]
def submit(book, message):
    remaining, d, p = message.size, message.direction, message.price
    fills = []
    while remaining > 0:
        best = book.best_price(-d)
        if best is None or best * d > p * d:
            break
        resting = book.size_at(-d, best)
        traded = min(remaining, resting)
        remaining -= traded
        book.set_size(-d, best, resting - traded)
        fills.append((best, traded))
    last = fills[-1][0] if fills else None
    rest = book.resting_price(p, last)
    if remaining > 0 and rest is not None:
        book.set_size(d, rest, book.size_at(d, rest) + remaining)
    return fills
\end{lstlisting}
\begin{lstlisting}[caption={Alternative B}, label=listing.precomputedMarketPartOnAWalkingSellB]
def submit(book, message):
    q, d, p = message.size, message.direction, message.price
    side = book.levels_map(-d)
    eligible = sum(s for price, s in side.items() if price * d <= p * d)
    q_m = min(q, eligible)
    fills, remaining = [], q_m
    while remaining > 0:
        best = book.best_price(-d)
        resting = book.size_at(-d, best)
        traded = min(remaining, resting)
        remaining -= traded
        book.set_size(-d, best, resting - traded)
        fills.append((best, traded))
    if q > q_m:
        book.set_size(d, p, book.size_at(d, p) + q - q_m)
    return fills
\end{lstlisting}
\begin{lstlisting}[caption={Run once with each alternative}, label=listing.precomputedMarketPartOnAWalkingSellRun]
book = AggregateBook.from_levels(
    {1000: 100, 999: 200, 998: 150}, {1002: 120, 1003: 180}
)
fills = submit(book, limit_order(1.0, 400, 999, SELL))
print(fills)
print(book.best_bid_price, book.best_ask_price, book.best_ask_size)
\end{lstlisting}
What happens?
\begin{choices}
 \correctchoice Both alternatives print \codehl{[(1000, 100), (999, 200)]},
  then \codehl{998 999 100}.
 \choice Alternative A prints \codehl{[(1000, 100), (999, 200)]}, then \codehl{998 999 100};
  Alternative B prints \codehl{[(1000, 100), (999, 200), (998, 100)]}, then \codehl{998 1002 120}.
 \choice Both alternatives print \codehl{[(999, 300)]}, then \codehl{998 999 100}.
 \choice Alternative A prints \codehl{[(1000, 100), (999, 200)]}, then \codehl{998 999 100};
  Alternative B raises \codehl{TypeError}.
\end{choices}
\explanation{The loop of A stops at 998, the first bid below the limit.
B stops after $\marketOrderSize = \min(400, 100 + 200) = 300$ shares,
all the size at eligible prices,
so its loop needs no price test and never meets an empty side.
Both rest $q - \marketOrderSize = 100$ at 999, which becomes the best ask,
and both price the fills at the resting orders.}
\end{question}

def submit(book, message):
    remaining, d, p = message.size, message.direction, message.price
    side = book.levels_map(-d)
    eligible = [price for price in side if price * d <= p * d]

    # ...
    levels = self.levels_map(direction)

        levels[price] = size
    else:
        levels.pop(price, None)

    if not levels:
        return None
    return max(levels) if direction == BUY else min(levels)

    direction = message.direction

\begin{question}
\label{question.sweepRescanningTheBestPrice}
Alternative~B below walks the eligible prices of the opposite side
in the order of priority of equation~\eqref{notes:eq.orderingOfLimitOrders},
sorted once,
where Alternative~A of Question~\ref{question.precomputedMarketPartOnAWalkingSell}
asks the book for its best price on every pass,
by the \codehl{best\_price} of Listing~\ref{listing.setSizeKeepingEmptyLevelsStorage}.
The two leave the same book and return the same fills.
They are timed on a market sell of $\ell$ shares
against a bid side of $\ell$ prices holding one share each,
the book built afresh before each run,
for $\ell = 10^{3}$ and $\ell = 10^{4}$.
\begin{lstlisting}[caption={Alternative B}, label=listing.sweepRescanningTheBestPriceB]
def submit(book, message):
    remaining, d, p = message.size, message.direction, message.price
    side = book.levels_map(-d)
    eligible = sorted(
        (price for price in side if price * d <= p * d),
        key=lambda price: price * d,
    )
    fills = []
    for best in eligible:
        if remaining == 0:
            break
        resting = side[best]
        traded = min(remaining, resting)
        remaining -= traded
        book.set_size(-d, best, resting - traded)
        fills.append((best, traded))
    last = fills[-1][0] if fills else None
    rest = book.resting_price(p, last)
    if remaining > 0 and rest is not None:
        book.set_size(d, rest, book.size_at(d, rest) + remaining)
    return fills
\end{lstlisting}
Which statement holds?
\begin{choices}
 \choice B is slower: it sorts the eligible prices of every order,
  where A finds each best price by a single scan.
 \choice Both times grow about tenfold, and A is slower by a constant factor.
 \correctchoice The time of A grows about a hundredfold, and that of B about tenfold.
 \choice Both times grow about a hundredfold.
\end{choices}
\explanation{A scans the keys of the bid side on each of its $\ell$ passes,
while the side holds $\ell, \ell - 1, \dots, 1$ prices:
of the order of $\ell^{2}/2$ comparisons.
B sorts once, in $O(\ell \log \ell)$ comparisons at most,
and then visits each price once.}
\end{question}

so that taking a share would empty the side.
\begin{lstlisting}[caption={\protect\codehl{unito26.lob.idealised.IdealisedFlow.\_place}}, label=listing.idealisedLimitBuyAtAFullTouchPlace]
def _place(self, book: AggregateBook, event: EventType, time: float) -> Message:
    direction = event.direction
    if event in (EventType.MARKET_BUY, EventType.MARKET_SELL):
        opposite = self._consumable_touch(book, -direction, time)
        return limit_order(time, 1, opposite, direction)
    if event in (EventType.WITHDRAW_BUY, EventType.WITHDRAW_SELL):
        own = self._consumable_touch(book, direction, time)
        return withdrawal(time, 1, own, direction)
    touch = book.best_price(direction)
    if book.size_at(direction, touch) < self.uniform_depth:
        return limit_order(time, 1, touch, direction)
    inside = touch + direction
    if inside == book.best_price(-direction):
        opposite = self._consumable_touch(book, -direction, time)
        return limit_order(time, 1, opposite, direction)
    return limit_order(time, 1, inside, direction)
\end{lstlisting}
