\begin{question}
\label{question.foldByReduceOverApply}
The fold~\eqref{notes:eq.bookFold} can be written with \codehl{reduce}, from \codehl{functools}:
\codehl{reduce(f, [y1, y2], x0)} returns \codehl{f(f(x0, y1), y2)}.
The method \codehl{apply} of the book processes one message in place,
and returns what \codehl{submit} or \codehl{withdraw} returns,
which is \codehl{None} when \codehl{record} is \codehl{False}.
The method \codehl{copy} is quoted below.
\begin{lstlisting}[caption={\protect\codehl{unito26.lob.orderbook.AggregateBook.copy}, its docstring elided.}, label=listing.foldByReduceOverApplyCopy]
def copy(self) -> "AggregateBook":
# ...
    clone = self._empty_like()
    for direction in (BUY, SELL):
        for price, size in self.levels_map(direction).items():
            clone.set_size(direction, price, size)
    return clone
\end{lstlisting}
\begin{lstlisting}[caption={Three folds}, label=listing.foldByReduceOverApplyFolds]
def fold_a(book, messages):
    for message in messages:
        book.apply(message, record=False)
    return book

def transition(book, message):
    after = book.copy()
    after.apply(message, record=False)
    return after

def fold_b(book, messages):
    return reduce(transition, messages, book)

def fold_c(book, messages):
    return reduce(lambda b, m: b.apply(m, record=False), messages, book)
\end{lstlisting}
\begin{lstlisting}[caption={Run once with each fold}, label=listing.foldByReduceOverApplyRun]
stream = [limit_order(1.0, 50, 1001, BUY), market_order(2.0, 80, SELL)]
book = AggregateBook.from_levels({1000: 100}, {1002: 120})
final = fold(book, stream)
print(final.best_bid_size, book.best_bid_size)
\end{lstlisting}
What does each fold do?
\begin{choices}
 \choice \codehl{fold\_a} and \codehl{fold\_b} print \codehl{70 70};
  \codehl{fold\_c} raises \codehl{AttributeError}.
 \choice \codehl{fold\_a} prints \codehl{70 70} and \codehl{fold\_b} prints \codehl{70 100};
  \codehl{fold\_c} prints \codehl{70 70}.
 \correctchoice \codehl{fold\_a} prints \codehl{70 70} and \codehl{fold\_b} prints \codehl{70 100};
  \codehl{fold\_c} raises \codehl{AttributeError}.
 \choice \codehl{fold\_a} and \codehl{fold\_b} print \codehl{70 100};
  \codehl{fold\_c} raises \codehl{AttributeError}.
\end{choices}
\explanation{The limit buy rests 50 at 1001,
and the market sell takes them and 30 of the 100 at 1000.
\codehl{fold\_a} changes the book it is given and returns it;
\codehl{fold\_b} changes only copies, and \codehl{book} keeps its opening 100.
In \codehl{fold\_c} the accumulator after the first message is the \codehl{None}
that \codehl{apply} returns, and the second step calls \codehl{apply} on it.}
\end{question}

\begin{question}
\label{question.copyingFoldAgainstInPlaceFold}
The folds \codehl{fold\_a} and \codehl{fold\_b}
of Listing~\ref{listing.foldByReduceOverApplyFolds}
are timed on one stream of 1000 messages:
limit orders and withdrawals of 10 shares at the five prices nearest each touch,
none of which crosses the spread.
Each fold starts from a book with $\ell$ occupied prices on each side,
for $\ell = 10$, $100$ and $1000$.
The book finds a best price by applying the builtin \codehl{max} or \codehl{min}
to the prices of a side.
Which statement holds?
\begin{choices}
 \choice The two take about the same time at every $\ell$:
  the messages reach only the five prices nearest each touch.
 \correctchoice The copying fold \codehl{fold\_b} is the slower,
  by a factor that grows with $\ell$ and exceeds ten at $\ell = 1000$.
 \choice The copying fold \codehl{fold\_b} is the slower,
  by a factor of about two at every $\ell$: it does the work of each message twice.
 \choice The two slow down by about the same factor from $\ell = 10$ to $\ell = 1000$,
  near the hundredfold growth of $\ell$,
  since each scans the $\ell$ prices of a side at every submission.
\end{choices}
\explanation{A copy calls the Python method \codehl{set\_size} once for each of the $2\ell$
occupied prices, at every message.
Both folds scan a side at every submission,
but inside a builtin, at a cost per price far below that of a method call,
so the copy dominates the time of \codehl{fold\_b} as $\ell$ grows.}
\end{question}

\begin{question}
\label{question.recursiveFoldOverAStream}
The fold~\eqref{notes:eq.bookFold} can also be written as a recursion,
as in \codehl{fold\_a} below.
The default limit on the depth of recursion in Python,
\codehl{sys.getrecursionlimit()}, is 1000 frames,
and a call that would exceed it raises \codehl{RecursionError}.
\begin{lstlisting}[caption={A recursive fold and a loop}, label=listing.recursiveFoldOverAStreamFolds]
def fold_a(book, messages):
    if not messages:
        return book
    book.apply(messages[0], record=False)
    return fold_a(book, messages[1:])

def fold_b(book, messages):
    for message in messages:
        book.apply(message, record=False)
    return book
\end{lstlisting}
\begin{lstlisting}[caption={Run with each fold, for $n = 500$ and for $n = 5000$}, label=listing.recursiveFoldOverAStreamRun]
stream = [limit_order(float(t), 10, 1000, BUY) for t in range(n)]
print(fold(AggregateBook(), stream).best_bid_size)
\end{lstlisting}
What happens?
\begin{choices}
 \choice Both folds print \codehl{5000} at $n = 500$ and \codehl{50000} at $n = 5000$.
 \choice \codehl{fold\_a} raises \codehl{RecursionError} at both lengths;
  \codehl{fold\_b} prints \codehl{5000}, then \codehl{50000}.
 \choice Both folds print \codehl{5000} at $n = 500$
  and raise \codehl{RecursionError} at $n = 5000$.
 \correctchoice \codehl{fold\_a} prints \codehl{5000} at $n = 500$
  and raises \codehl{RecursionError} at $n = 5000$;
  \codehl{fold\_b} prints \codehl{5000}, then \codehl{50000}.
\end{choices}
\explanation{Python keeps the frame of a call in tail position,
so \codehl{fold\_a} holds $n+1$ frames of its own at its deepest;
\codehl{apply} and the methods it calls return before the next message is taken,
and add a few frames at a time, not a few per message.
\codehl{fold\_b} keeps one frame for any length,
and $n$ orders of 10 shares at one price rest as $10n$ at 1000.}
\end{question}

Two books are compared by folding the same stream through each,

def run(book: AggregateBook, messages: Iterable[Message]) -> Counter:

\begin{question}

The spread after every message of a stream can be read off one fold,

        run(book, messages[:k])
        spreads.append(book.spread)

def spreads_b(messages):
    book, spreads = AggregateBook(), []
    for message in messages:
        book.apply(message, record=False)
        spreads.append(book.spread)
    return spreads
\end{lstlisting}

\begin{choices}
 \choice The two lists are equal, and \codehl{spreads\_a} is the slower
  by a factor about the same at every $n$.
 \choice \codehl{spreads\_a} is the faster:

 \choice The two lists differ by one position:
  the first entry of \codehl{spreads\_a} is the spread of the empty book.

\begin{question}
\label{question.deltaLogReplayIsIdempotent}
\codehl{DeltaLog.record}, of \codehl{unito26.lob.delta\_log}, quoted below,
folds a stream through a book
and keeps, for each message, the level changes it made:
each a \codehl{LevelDelta} carrying the side, the price,
and the new size resting there, \codehl{resting}, which is zero when the price empties.
\codehl{MarketSession.from\_delta\_log} replays them through \codehl{set\_size},
as \codehl{replay\_deltas} does below.
A feed may deliver a record twice, or lose one.
\begin{lstlisting}[caption={\protect\codehl{unito26.lob.delta\_log.DeltaLog.record}}, label=listing.deltaLogReplayIsIdempotentRecord]
@classmethod
def record(cls, book: AggregateBook, messages: Iterable[Message]) -> "DeltaLog":
    opening_bids = dict(book.levels_map(BUY))
    opening_asks = dict(book.levels_map(SELL))
    times: list[float] = []
    entries: list[tuple[int, float, LevelDelta]] = []
    for sequence, message in enumerate(messages):
        result = book.apply(message, record=True)
        times.append(message.time)
        entries += [(sequence, message.time, delta) for delta in result.deltas]
    return cls(opening_bids, opening_asks, times, entries)
\end{lstlisting}
\begin{lstlisting}[caption={A repeated and a lost record}, label=listing.deltaLogReplayIsIdempotentReplays]

deltas = [delta for _, _, delta in log.entries]

def replay_deltas(deltas):

    return book.bids

clean = replay_messages(stream)
print(replay_messages(stream[:1] + stream) == clean,
      replay_deltas(deltas[:1] + deltas) == clean,
      replay_deltas(deltas[1:]) == clean,
      replay_deltas(deltas[:-1]) == clean)
\end{lstlisting}

 \choice \codehl{False False False False}
 \choice \codehl{False True False False}
 \choice \codehl{False True True True}
\end{choices}
\explanation{The log holds three bid deltas: 150 at 1000, then 30 at 1000, then 160 at 999.
A delta writes an absolute size, so a repeat delivered next to the original changes nothing,
